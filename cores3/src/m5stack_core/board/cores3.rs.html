2//! M5Stack CoreS3 board-level bring-up sequences and pin wiring.
3//!
4//! # Internal I²C bus (`I2C_SYS`)
5//!
6//! One shared bus on **GPIO12 (SDA) / GPIO11 (SCL)** carries seven devices; the
7//! BSP drives the first three, the rest are listed for completeness:
8//!
9//! | Device   | Addr | Role                        | BSP driver          |
10//! |----------|------|-----------------------------|---------------------|
11//! | AXP2101  | 0x34 | PMIC / power rails / battery | [`Axp2101Driver`]   |
12//! | AW9523B  | 0x58 | IO expander (LCD/touch reset)| [`Aw9523bDriver`]   |
13//! | FT6336U  | 0x38 | capacitive touch            | `driver::ft6336u`   |
14//! | BMI270   | 0x69 | 6-axis IMU                  | —                   |
15//! | BM8563   | 0x51 | RTC                         | —                   |
16//! | ES7210   | 0x40 | audio ADC (mic)             | —                   |
17//! | AW88298  | 0x36 | speaker amp                 | —                   |
18//!
19//! # AXP2101 power rails (from `Sch_M5_CoreS3_v1.0`)
20//!
21//! | AXP2101 out | Net           | Spec        | Powers                                  |
22//! |-------------|---------------|-------------|-----------------------------------------|
23//! | DCDC1       | `VDD_3V3`     | 3.3 V / 2 A | **Core VDD** — ESP32-S3, **AW9523B**, **FT6336U touch**, LCD logic |
24//! | DCDC3       | `VCC_3V3`     | 3.3 V / 2 A | Peripherals VDD                         |
25//! | DLDO1       | `VCC_BL`      | 3.3 V       | LCD **backlight** (see [`BACKLIGHT_MV`]) |
26//! | ALDO1       | `VDD_1V8`     | 1.8 V/300 mA| AW88298 PA DVDD                         |
27//! | ALDO2       | `VDDA_3V3`    | 3.3 V/300 mA| ES7210 codec VDDP                       |
28//! | ALDO3       | `VDDCAM_3V3`  | 3.3 V/300 mA| camera / codec VDDA                     |
29//! | ALDO4       | `VDD_3V3_SD`  | 3.3 V/300 mA| microSD card VDD                        |
30//! | BLDO1       | `AVDD`        | 2.8 V/300 mA| sensor VDDA                             |
31//! | BLDO2       | `DVDD`        | 1.2 V/300 mA| sensor VDD                              |
32//!
33//! **Reset gating (important):** the **AW9523B `RSTN` is wired to `AXP_PG`**
34//! (AXP2101 power-good) and has an internal ~100 kΩ pull-*low* — so the expander
35//! is held in reset until the AXP asserts power-good. It in turn drives
36//! `LCD_RST` / `TOUCH_RST`, so panel + touch bring-up all chain off it. See
37//! [`power_display_reset`] for the cold-boot readiness handling.
38

60/// AXP2101 PMIC I2C address on CoreS3.
61const AXP2101_ADDR: u8 = 0x34;
62/// Display backlight rail: AXP2101 DLDO1 @ 3.3 V.
63const BACKLIGHT_MV: u16 = 3300;
64
65/// AW9523B post-power-on I²C-ready time — the datasheet's "minimal wait time for
66/// I2C communication is 5 ms".
67const AW9523B_POR_MS: u64 = 5;
68/// Deterministic settle before the first AW9523B access: 200% of the POR time.
69/// On CoreS3 the expander's `RSTN` has an internal ~100 kΩ pull-*low* and is
70/// wired to `AXP_PG` (AXP2101 power-good), so it is held in reset until the AXP
71/// asserts power-good; at cold power-on `VDD_3V3` is already up (it powers the
72/// running ESP32) but `AXP_PG` may lag, leaving the expander briefly in reset.
73/// Waiting first — rather than poking the bus and retrying — avoids the premature
74/// `AcknowledgeCheckFailed(Address)`, each of which fires one chip-wide I2C clock
75/// reset.
76const AW9523B_SETTLE_MS: u64 = 2 * AW9523B_POR_MS;
77/// Fallback if the expander is still not ready after the settle. This should not
78/// happen in normal operation, so each attempt logs a warning; bounded, with a
79/// final failure logged as an error (best-effort — bring-up continues).
80const AW9523B_FALLBACK_TRIES: u32 = 5;
81const AW9523B_FALLBACK_DELAY_MS: u64 = 5;
82
83/// Fault-injection predicate for the `test-fault-inject` feature: reports the
84/// AW9523B "not ready" for the first `M5_TEST_AW_NACK` post-settle attempts (a
85/// build-time env var), so the cold-boot fallback retry can be regression-tested
86/// on the bench without a real power cycle. Absent from production builds.
87#[cfg(feature = "test-fault-inject")]
88fn aw_fault_inject_active(tries: u32) -> bool {
89    match option_env!("M5_TEST_AW_NACK") {
90        Some(s) => tries < s.parse::<u32>().unwrap_or(0),
91        None => false,
92    }
93}

95/// Power + display bring-up over the shared I2C bus: AW9523B IO-expander
96/// (`LCD_RST` + touch reset) then AXP2101 PMIC backlight (DLDO1 @ 3.3 V).
97///
98/// A [`AW9523B_SETTLE_MS`] wait precedes the first AW9523B access so the expander
99/// is out of reset (its `RSTN` follows `AXP_PG`) before we touch the bus — this
100/// removes the intermittent cold-boot `AcknowledgeCheckFailed(Address)` that
101/// otherwise fired one chip-wide I2C clock reset at every power-on. A bounded
102/// warn-on-failure retry backs it up in case the settle was not enough.
103///
104/// Best-effort — each sub-step logs and continues on error, so a flaky chip
105/// can't block the display/control loop from coming up. Caller-driven so the
106/// caller owns the *when* and the *which core*: this MUST run on the core that
107/// owns the I2C IRQ (APP core on the regulator — see its `irq-core-binding`
108/// doc), and the caller signals display-readiness after it returns
109/// (`LCD_RST` must precede display init).
110pub async fn power_display_reset(i2c: &'static SharedI2cBus) {
111    let mut aw = Aw9523bDriver::new(Aw9523bResources { i2c });
112    // Deterministic settle for the cold-boot reset-release window (RSTN = AXP_PG)
113    // before the first bus access — see AW9523B_SETTLE_MS.
114    Timer::after(Duration::from_millis(AW9523B_SETTLE_MS)).await;
115    let mut tries = 0u32;
116    loop {
117        // Test hook (feature `test-fault-inject`, absent in production builds):
118        // simulate the AW9523B still being not-ready for the first
119        // `M5_TEST_AW_NACK` post-settle attempts, so the fallback retry is
120        // exercisable on the bench without a power cycle (which the rig can't do).
121        #[cfg(feature = "test-fault-inject")]
122        if aw_fault_inject_active(tries) {
123            tries += 1;
124            if tries > AW9523B_FALLBACK_TRIES {
125                error!(
126                    "AW9523B not ready (fault-inject) after {} retries",
127                    AW9523B_FALLBACK_TRIES
128                );
129                break;
130            }
131            warn!(
132                "[test-fault-inject] AW9523B not-ready simulated; retry {}/{}",
133                tries, AW9523B_FALLBACK_TRIES
134            );
135            Timer::after(Duration::from_millis(AW9523B_FALLBACK_DELAY_MS)).await;
136            continue;
137        }
138        match aw.init().await {
139            Ok(()) => {
140                if tries > 0 {
141                    info!("AW9523B ready after settle + {} retries", tries);
142                }
143                break;
144            }
145            Err(e) => {
146                tries += 1;
147                if tries > AW9523B_FALLBACK_TRIES {
148                    error!(
149                        "AW9523B not ready after {} ms settle + {} retries: {:?}",
150                        AW9523B_SETTLE_MS, AW9523B_FALLBACK_TRIES, e
151                    );
152                    break;
153                }
154                warn!(
155                    "AW9523B not ready after settle; retry {}/{}: {:?}",
156                    tries, AW9523B_FALLBACK_TRIES, e
157                );
158                Timer::after(Duration::from_millis(AW9523B_FALLBACK_DELAY_MS)).await;
159            }
160        }
161    }
162    if let Err(e) = aw.lcd_rst_pulse().await {
163        error!("AW9523B LCD RST failed: {:?}", e);
164    }
165    if let Err(e) = aw.touch_rst_pulse().await {
166        error!("AW9523B TOUCH RST failed: {:?}", e);
167    }
168    let mut axp = Axp2101Driver::new(i2c, AXP2101_ADDR);
169    if let Err(e) = axp.set_dldo1(true, BACKLIGHT_MV).await {
170        error!("AXP2101 backlight enable failed: {:?}", e);
171    }
172}

174// --- GPIO35 MISO/DC mux -----------------------------------------------------
175
176/// GPIO35 bit position in the GPIO1 (pins 32–) register bank.
177const GPIO35_BIT: u32 = 1 << (35 - 32);
178
179/// Zero-sized display-DC "pin" that drives GPIO35 via direct register writes.
180///
181/// On the CoreS3, GPIO35 is hardware-shared between SPI2 MISO (SD-card reads)
182/// and display DC (command/data select). This type does NOT own the pin — the
183/// SPI peripheral owns it for MISO input (`with_miso`), which also routes the
184/// pad through the GPIO matrix. Each `set_low`/`set_high` re-enables the
185/// output driver, so it composes with [`gpio35_disable_output`] (called after
186/// every SD access window).
187///
188/// Two valid configurations for GPIO35 as DC:
189/// 1. **No SD use** (simple display apps): configure GPIO35 as a plain
190///    [`Output`](esp_hal::gpio::Output) instead — a bare register hack without
191///    any pad routing leaves the pad unrouted and DC never toggles.
192/// 2. **Display + SD shared** ([`super::spi2`]): SPI owns the pad as MISO and
193///    DC writes go through this type.
194///
195/// Safety of the mux: the display and the SD card must share one SPI-bus
196/// mutex within a single task, with no `.await` between a DC write and the
197/// SPI transfer — otherwise an SD operation could preempt mid-command.
198pub struct Gpio35Dc;
199
200impl ErrorType for Gpio35Dc {
201    type Error = core::convert::Infallible;
202}
203
204/// Pending DC level, applied by [`apply_pending_dc`] when the display actually
205/// owns the bus.
206static DC_LOW: core::sync::atomic::AtomicBool = core::sync::atomic::AtomicBool::new(false);
207
208impl OutputPin for Gpio35Dc {
209    // RECORD, do not drive.
210    //
211    // GPIO35 is MISO as well as DC. `SpiInterface` sets DC and *then* awaits the
212    // SPI transfer, and that await now blocks on the shared-bus permit — so a
213    // DC write that took effect immediately would hold GPIO35 as an output
214    // while the SD card is mid-command, and every card response would read back
215    // as 0x00. (Observed exactly so: "unexpected token 0x00",
216    // "CrcMismatch(0, ..)", "UnknownDataResponse(0)".)
217    //
218    // The module docs used to require "no `.await` between a DC write and the
219    // SPI transfer". The fair arbiter introduced one, so the requirement is
220    // enforced here instead of being documented and hoped for: the level is
221    // recorded now and applied in [`apply_pending_dc`] once the permit is held.
222    fn set_low(&mut self) -> Result<(), Self::Error> {
223        DC_LOW.store(true, core::sync::atomic::Ordering::Release);
224        Ok(())
225    }
226
227    fn set_high(&mut self) -> Result<(), Self::Error> {
228        DC_LOW.store(false, core::sync::atomic::Ordering::Release);
229        Ok(())
230    }
231}
232
233/// Drive GPIO35 to the DC level recorded by [`Gpio35Dc`].
234///
235/// Call ONLY while holding the SPI2 permit: it takes the pad away from MISO.
236pub fn apply_pending_dc() {
237    unsafe {
238        let gpio = &*esp_hal::peripherals::GPIO::PTR;
239        if DC_LOW.load(core::sync::atomic::Ordering::Acquire) {
240            gpio.out1_w1tc().write(|w| w.bits(GPIO35_BIT));
241        } else {
242            gpio.out1_w1ts().write(|w| w.bits(GPIO35_BIT));
243        }
244        gpio.enable1_w1ts().write(|w| w.bits(GPIO35_BIT));
245    }
246}
247
248/// Disable the GPIO35 output driver so the pad returns to high-impedance
249/// input (SPI2 MISO). Must be called before any SD-card SPI operation — both
250/// at bring-up (after display init; [`super::spi2`] does it) and per-op at
251/// runtime (pass this as the block-device handler's pre-op hook).
252pub fn gpio35_disable_output() {
253    unsafe {
254        let gpio = &*esp_hal::peripherals::GPIO::PTR;
255        gpio.enable1_w1tc().write(|w| w.bits(GPIO35_BIT));
256    }
257}

261/// External M5-Bus pins the BSP does not claim for on-board functions. What
262/// hangs off them (1-Wire sensors, RPM pickups, ...) is application wiring.
263/// Not exhaustive — extend as needed.
264pub struct M5Bus<'a> {
265    /// M5-Bus pin 20 (`BUS_PA_SCL`) → GPIO1.
266    pub gpio1: AnyPin<'a>,
267    /// M5-Bus pin 10 (`BUS_PB_OUT`) → GPIO9.
268    pub gpio9: AnyPin<'a>,
269    /// M5-Bus pin 2 (`BUS_ADC1`) → GPIO10.
270    pub gpio10: AnyPin<'a>,
271    /// M5-Bus pin 8 (`BUS_G5`) → GPIO5.
272    pub gpio5: AnyPin<'a>,
273    /// M5-Bus pin 21 (`BUS_G6`) → GPIO6.
274    pub gpio6: AnyPin<'a>,
275    /// M5-Bus pin 22 (`BUS_G7`) → GPIO7.
276    pub gpio7: AnyPin<'a>,
277    /// M5-Bus pin 13 (`BUS_U0RXD`) → GPIO44 — the ROM console's RX pin,
278    /// unused on CoreS3 (console is USB-Serial-JTAG). Assign as an input:
279    /// a ROM boot-log burst on the TX side (`gpio43`) can then never
280    /// contend with a driven line here.
281    pub gpio44: AnyPin<'a>,
282    /// M5-Bus pin 14 (`BUS_U0TXD`) → GPIO43 — the ROM console's TX pin,
283    /// unused on CoreS3. Assign as an output; see [`Self::gpio44`].
284    pub gpio43: AnyPin<'a>,
285}

287/// The CoreS3's peripherals, split into board-fact groups. See
288/// [`Board::split`].
289pub struct Board {
290    /// Display + SD-card shared SPI2 bus (see [`super::spi2`]).
291    pub spi2: Spi2Resources<'static>,
292    /// Internal I2C bus (SDA=GPIO12, SCL=GPIO11) shared by the AXP2101 PMIC
293    /// (0x34), AW9523B expander (0x58) and FT6336U touch (0x38).
294    ///
295    /// Returned BLOCKING: defer `into_async()` to the core that runs the I2C
296    /// tasks — `into_async()` binds the I2C peripheral IRQ to the calling
297    /// core, and on a BLE app it must stay off the core running RWBLE's
298    /// level-1 IRQ (cumulative IRQ blocking otherwise desyncs the BLE
299    /// controller blob and wedges that core's executor).
300    pub i2c0: I2c<'static, Blocking>,
301    pub bt: BT<'static>,
302    pub wifi: WIFI<'static>,
303    /// USB-Serial-JTAG — console + serial-cmd
304    /// ([`crate::io::serial_cmd::SerialCmdResources`]).
305    pub usb_device: USB_DEVICE<'static>,
306    /// RMT — e.g. the 1-Wire driver ([`crate::io::ow_temp::OnewireResources`]).
307    pub rmt: RMT<'static>,
308    /// PCNT — e.g. pulse counting ([`crate::io::rpm::RpmResources`]).
309    pub pcnt: PCNT<'static>,
310    /// Spare SPI3 unit — the BSP does not drive it, listed for the app to
311    /// claim (same idea as [`Self::rmt`]/[`Self::pcnt`]). Wiring (which
312    /// [`Self::m5bus`] pin is SCK/MOSI/MISO/CS) and the interrupt priority /
313    /// core allocation are application knowledge. Attach the CS pin as the
314    /// peripheral's **hardware** chip-select (`with_cs`), not a
315    /// software-toggled [`esp_hal::gpio::Output`] — some designs depend on a
316    /// sub-microsecond CS gap that software toggling cannot hold. See #27.
317    pub spi3: AnySpi<'static>,
318    /// GDMA channel for [`Self::spi3`] — any free channel works, this is just
319    /// not the one `spi2` already owns (`DMA_CH0`).
320    pub spi3_dma: AnyGdmaChannel<'static>,
321    /// SK6812 LED bars data pin (M-Bus pin 23 → GPIO13 on CoreS3), driven over
322    /// RMT ([`crate::driver::sk6812::Sk6812Driver`]). The M5GO bottom's 5 V LED
323    /// rail is gated by the AW9523B — see `Aw9523bDriver::enable_bus_5v`.
324    pub sk6812: AnyPin<'static>,
325    /// External SPI PSRAM (~8 MB) — feed to `mem::psram_map` / `mem::psram_split`
326    /// (sizing/usage is application policy). [feature `psram`]
327    pub psram: esp_hal::peripherals::PSRAM<'static>,
328    pub m5bus: M5Bus<'static>,
329    pub system: SystemResources<'static>,
330}
331
332impl Board {
333    /// Wire the CoreS3 pin map. Call once with the output of
334    /// [`crate::board::init`] (or `esp_hal::init`); heap setup stays with the
335    /// app (sizing is application policy).
336    pub fn split(peripherals: Peripherals) -> Self {
337        let spi2 = Spi2Resources {
338            spi2: AnySpi::from(peripherals.SPI2),
339            spi2_dma: peripherals.DMA_CH0.degrade(),
340            sck: AnyPin::from(peripherals.GPIO36),
341            mosi: AnyPin::from(peripherals.GPIO37),
342            miso_dc: AnyPin::from(peripherals.GPIO35),
343            display_cs: AnyPin::from(peripherals.GPIO3),
344            card_cs: AnyPin::from(peripherals.GPIO4),
345        };
346
347        let i2c0 = I2c::new(
348            peripherals.I2C0,
349            I2cConfig::default()
350                .with_frequency(Rate::from_khz(400))
351                .with_timeout(BusTimeout::BusCycles(20))
352                // Bound async transactions: without a software timeout the
353                // deadline is None, so a stuck/NACKing transaction loops
354                // forever in esp-hal's `check_all_commands_done` yield_now
355                // spin. On an InterruptExecutor that spin pins the core,
356                // masking the very completion IRQ that would end it — the
357                // whole executor wedges (HIL-proven on S3, 2026-06-02). A
358                // finite deadline makes a stuck op return Err instead.
359                .with_software_timeout(SoftwareTimeout::Transaction(
360                    esp_hal::time::Duration::from_millis(25),
361                )),
362        )
363        .expect("I2C0 init failed")
364        .with_sda(AnyPin::from(peripherals.GPIO12))
365        .with_scl(AnyPin::from(peripherals.GPIO11));
366
367        let tg0 = TimerGroup::new(peripherals.TIMG0);
368        let tg1 = TimerGroup::new(peripherals.TIMG1);
369        let system = SystemResources {
370            sw_int: SoftwareInterruptControl::new(peripherals.SW_INTERRUPT),
371            timer0_0: AnyTimer::from(tg0.timer0),
372            timer0_1: AnyTimer::from(tg0.timer1),
373            timer1_0: AnyTimer::from(tg1.timer0),
374            timer1_1: AnyTimer::from(tg1.timer1),
375            cpu_ctrl: peripherals.CPU_CTRL,
376            lpwr: peripherals.LPWR,
377        };
378
379        Board {
380            spi2,
381            i2c0,
382            bt: peripherals.BT,
383            wifi: peripherals.WIFI,
384            usb_device: peripherals.USB_DEVICE,
385            rmt: peripherals.RMT,
386            pcnt: peripherals.PCNT,
387            spi3: AnySpi::from(peripherals.SPI3),
388            spi3_dma: peripherals.DMA_CH1.degrade(),
389            sk6812: AnyPin::from(peripherals.GPIO13),
390            psram: peripherals.PSRAM,
391            m5bus: M5Bus {
392                gpio1: AnyPin::from(peripherals.GPIO1),
393                gpio9: AnyPin::from(peripherals.GPIO9),
394                gpio10: AnyPin::from(peripherals.GPIO10),
395                gpio5: AnyPin::from(peripherals.GPIO5),
396                gpio6: AnyPin::from(peripherals.GPIO6),
397                gpio7: AnyPin::from(peripherals.GPIO7),
398                gpio44: AnyPin::from(peripherals.GPIO44),
399                gpio43: AnyPin::from(peripherals.GPIO43),
400            },
401            system,
402        }
403    }
