2//! BLE radio driver wrapper around `esp-radio`.
3//!
4//! Initializes the ESP32/ESP32-S3 radio coprocessor and returns a
5//! `BleConnector` for use with the `trouble-host` BLE stack.
6//!
7//! Ref: <https://docs.espressif.com/projects/esp-idf/en/stable/esp32/api-reference/bluetooth/index.html>
8use esp_hal::peripherals::BT;
9use esp_radio::ble::{
10    Config, InvalidConfigError,
11    controller::{BleConnector, BleInitError},
12};
13use thiserror_no_std::Error;
14
15#[derive(Debug, Error)]
16pub enum RadioError {
17    #[error("Failed to initialize BLE controller")]
18    BleInitError(#[from] BleInitError),
19
20    #[error("Failed to initialize BLE controller")]
21    BleConfigError(#[from] InvalidConfigError),
22}
23
24pub struct BleRadio {
25    pub ble_connector: BleConnector<'static>,
26}
27
28impl BleRadio {
29    /// Initialize BLE in scan-only mode with minimal buffer footprint.
30    ///
31    /// esp-radio's `Config::default()` sizes the BT-controller blob for
32    /// `max_connections=6` and `scan_duplicate_list_count=100`. On ESP32-S3
33    /// that defaults to ~36 KB of heap, which doesn't fit alongside LVGL on
34    /// cores3 (56 KB heap). Our use case is passive scanning of Victron
35    /// advertisements — we never open a connection, and only a handful of
36    /// distinct advertisers exist on the bus.
37    ///
38    /// `max_connections=1` is the minimum the controller accepts. Setting
39    /// it to 1 shrinks the per-connection static state. Combined with a
40    /// 10-entry duplicate-filter (the spec minimum, range 10–1000), this
41    /// frees ~10 KB on ESP32-S3 and a smaller amount on ESP32.
42    pub fn new(bt: BT<'static>) -> Result<Self, RadioError> {
43        // task_stack_size: BLE controller's own task stack (esp-radio default
44        // is 4096 B). The HIL hit a stack-guard watchpoint in
45        // `ld_sscan_frm_cbk` (controller's scan-frame callback): SP within
46        // 32 B of the per-task guard while a `r_rwbtdm_isr_wrapper` was
47        // nesting on top. Bumping to 8 KB gives the controller's deepest
48        // ISR-nested path room without resizing the main task stack region
49        // (which is pinned at [28, 80] KB by ESP_HAL_CONFIG_MAIN_STACK_*).
50        // Costs 4 KB of heap — well within budget on both targets.
51        let cfg = Config::default()
52            .with_task_priority(10)
53            .with_max_connections(1)
54            .with_scan_duplicate_list_count(10)
55            .with_task_stack_size(8192);
56        let ble_connector = BleConnector::new(bt, cfg)?;
57
58        Ok(Self { ble_connector })
59    }
60}
