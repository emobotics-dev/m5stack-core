2//! Async buffered serial console for both targets — byte-level ring buffer
3//! with **overwrite-on-full** semantics. The producer (`log!()` / [`send_line`])
4//! is O(1) and NEVER blocks the caller: never spins, never awaits, never
5//! creates back-pressure. On full, the **oldest** bytes are overwritten so the
6//! lines fired *just before* a failure survive (those are the informative ones).
7//!
8//! A single [`drain_task`] pulls contiguous slices from the ring and writes
9//! them via the target's async TX sink (`write_all().await` parks on the
10//! TX-done IRQ when the FIFO is full — no busy-spin). Producers signal the
11//! drain after every write; the drain awaits the signal when the ring is empty.
12//!
13//! Per-target seam (hardware): the sink types + [`setup`] (build + split the
14//! peripheral) + [`imp::boot_panic_write`]. fire27 = UART0 @ 1 Mbaud; cores3 =
15//! USB-Serial-JTAG CDC. `setup` does NOT make the fire27 TX async — `into_async()`
16//! binds the IRQ to the *calling* core, so the binary does it from `main` (PRO).
17//!
18//! The firmware's `alternator_regulator::logger::cat_line` calls [`send_line`]
19//! for the `:cat` dump. Unlike `log!()`, [`send_line`] is **back-pressuring**:
20//! it awaits ring space before writing, so a fast read-back self-paces to the TX
21//! drain rate and is lossless (a plain overwrite-on-full write dropped lines and
22//! made `log_interval`'s read-back show false gaps). alternator-regulator depends
23//! on this crate ONLY for that — optional + esp-hal-gated, so host builds never
24//! pull it.

84    // Raw UART0 TX-FIFO writer (panic only). Spins for FIFO room with a
85    // bounded budget per byte: the panic message MUST get out — a dropped
86    // [PANIC] line turns a clean panic into a silent "wedge" (cost a long
87    // stack-overflow hunt on cores3). The bound keeps a dead/unclocked UART
88    // from hanging the panic loop forever. Used by `on_panic` to synchronously
89    // flush the ring after the async drain is gone (or never started). NEVER
90    // call from steady-state code — `log!()` / `send_line` go through the ring.
91    const UART0_FIFO_REG: *mut u32 = 0x3FF4_0000 as *mut u32;
92    const UART0_STATUS_REG: *const u32 = 0x3FF4_001C as *const u32;
93    const TX_FIFO_DEPTH: u32 = 128;
94    /// ~a few ms at CPU speed — plenty for one byte at 1 Mbaud.
95    const PANIC_SPIN_PER_BYTE: u32 = 1_000_000;
96
97    pub fn boot_panic_write(bytes: &[u8]) {
98        for &b in bytes {
99            let mut budget = PANIC_SPIN_PER_BYTE;
100            while unsafe { (UART0_STATUS_REG.read_volatile() >> 16) & 0xFF } >= TX_FIFO_DEPTH - 2 {
101                budget -= 1;
102                if budget == 0 {
103                    return; // UART dead — give up rather than hang the panic
104                }
105                core::hint::spin_loop();
106            }
107            unsafe { UART0_FIFO_REG.write_volatile(b as u32) };
108        }
109    }
110}

144    // Raw SERIAL_JTAG EP1 FIFO writer (panic only). Spins for FIFO room with
145    // a bounded budget per fill: the panic message MUST get out — the old
146    // drop-on-full policy lost the [PANIC] line whenever the ring held more
147    // than one 64-byte EP buffer of pre-panic context, turning every panic
148    // into a silent "wedge" (cost a long stack-overflow hunt). The bound
149    // keeps an unplugged USB host from hanging the panic loop forever.
150    const SERIAL_JTAG_FIFO_REG: *mut u32 = 0x6003_8000 as *mut u32;
151    const SERIAL_JTAG_CONF_REG: *mut u32 = 0x6003_8004 as *mut u32;
152    /// ~tens of ms at CPU speed — the host polls the 64-byte EP every USB
153    /// micro-frame, so a live host clears the FIFO well within this.
154    const PANIC_SPIN_PER_FILL: u32 = 5_000_000;
155
156    #[inline]
157    fn fifo_full() -> bool {
158        unsafe { SERIAL_JTAG_CONF_REG.read_volatile() & 0b010 == 0 }
159    }
160
161    pub fn boot_panic_write(bytes: &[u8]) {
162        for &b in bytes {
163            if fifo_full() {
164                // Hand the queued bytes to the host, then wait (bounded) for
165                // room. No host within the budget → give up, don't hang.
166                unsafe { SERIAL_JTAG_CONF_REG.write_volatile(0b001) }; // flush (wr_done)
167                let mut budget = PANIC_SPIN_PER_FILL;
168                while fifo_full() {
169                    budget -= 1;
170                    if budget == 0 {
171                        return;
172                    }
173                    core::hint::spin_loop();
174                }
175            }
176            unsafe { SERIAL_JTAG_FIFO_REG.write_volatile(b as u32) };
177        }
178        unsafe { SERIAL_JTAG_CONF_REG.write_volatile(0b001) }; // flush (wr_done)
179    }
180}

192// ---- byte-level ring buffer (target-agnostic) ----
193
194/// Ring capacity. ~50 lines × 80 B = ~4 KB — same memory budget as the prior
195/// `Channel<Line, 12>` (~4.2 KB), and large enough to hold a message-only
196/// panic plus the immediate pre-failure context. Bump if `esp-backtrace` is
197/// ever wired up (a trace would add several KB).
198const RING_SIZE: usize = 4096;
199/// Per-line stack-format buffer. Largest record is the `[hil-cat]` CSV dump
200/// (≈ 320 B + prefix + CRLF).
201const LINE_CAP: usize = 352;
202
203/// Byte ring with overwrite-on-full. Single struct held inside the mutex.
204struct Ring {
205    buf: [u8; RING_SIZE],
206    head: usize, // write position (next byte to be written)
207    tail: usize, // read position (next byte to be read)
208    full: bool,  // disambiguates empty vs full when head == tail
209    /// Bytes silently overwritten (oldest discarded) since the last drain read.
210    /// The drain turns this into a VISIBLE `[CONSOLE-DROP …]` marker so a
211    /// ring overrun never looks like a clean gap (which previously made
212    /// `log_interval`'s read-back fail mysteriously). Saturates.
213    dropped: u32,
214}
215
216impl Ring {
217    const fn new() -> Self {
218        Self {
219            buf: [0; RING_SIZE],
220            head: 0,
221            tail: 0,
222            full: false,
223            dropped: 0,
224        }
225    }
226
227    /// Append `bytes`; on full, the **oldest** bytes are overwritten. Always
228    /// succeeds — no return value, no error path, no waiting. Called inside
229    /// the mutex by both the log macros and `send_line`. Each overwritten byte
230    /// bumps `dropped` so the drain can flag the loss.
231    fn write(&mut self, bytes: &[u8]) {
232        for &b in bytes {
233            self.buf[self.head] = b;
234            self.head = (self.head + 1) % RING_SIZE;
235            if self.full {
236                // Overwrote the tail byte — advance tail to track it, and
237                // record the drop so it surfaces downstream.
238                self.tail = (self.tail + 1) % RING_SIZE;
239                self.dropped = self.dropped.saturating_add(1);
240            } else if self.head == self.tail {
241                self.full = true;
242            }
243        }
244    }
245
246    /// Read + reset the overwritten-byte counter (drain-side).
247    #[cfg(feature = "console-serial")]
248    fn take_dropped(&mut self) -> u32 {
249        core::mem::take(&mut self.dropped)
250    }
251
252    #[cfg(feature = "console-serial")]
253    fn is_empty(&self) -> bool {
254        !self.full && self.head == self.tail
255    }
256
257    /// Bytes currently queued (unread).
258    fn used(&self) -> usize {
259        if self.full {
260            RING_SIZE
261        } else if self.head >= self.tail {
262            self.head - self.tail
263        } else {
264            RING_SIZE - self.tail + self.head
265        }
266    }
267
268    /// Free bytes available before overwrite-on-full would discard data.
269    fn free(&self) -> usize {
270        RING_SIZE - self.used()
271    }
272
273    /// Copy up to `dst.len()` readable bytes into `dst` and advance `tail`.
274    /// Bytes are taken from a single contiguous slice — if the ring wraps,
275    /// the caller will get the rest on the next call. Returns bytes copied.
276    #[cfg(feature = "console-serial")]
277    fn read_and_consume(&mut self, dst: &mut [u8]) -> usize {
278        if self.is_empty() {
279            return 0;
280        }
281        // Length of the next contiguous readable slice.
282        let n_contig = if self.head > self.tail {
283            self.head - self.tail
284        } else {
285            RING_SIZE - self.tail
286        };
287        let n = n_contig.min(dst.len());
288        let end = self.tail + n;
289        dst[..n].copy_from_slice(&self.buf[self.tail..end]);
290        self.tail = end % RING_SIZE;
291        if n > 0 {
292            self.full = false;
293        }
294        n
295    }
296}

298/// The ring + a CriticalSectionRawMutex so any task on any core can write
299/// safely. Lock scope is per-op (one memcpy + a few indices) — never held
300/// across an `.await`, never spans more than one log line.
301static RING: BlockingMutex<CriticalSectionRawMutex, RefCell<Ring>> =
302    BlockingMutex::new(RefCell::new(Ring::new()));
303
304/// Producer signal — woken after every ring write so the drain can resume
305/// when the ring was empty. Idempotent (multiple signals before a wait =
306/// one wait wakes); the drain reads everything it can per wakeup.
307static DRAIN_SIGNAL: Signal<CriticalSectionRawMutex, ()> = Signal::new();
308
309/// Drain → producer signal — raised after every drain read so a *back-pressuring*
310/// producer ([`send_line`], the HIL `:cat` dump) can wake and retry once the ring
311/// has freed space. The non-blocking hot log path ([`push_line`]) ignores it.
312static SPACE_SIGNAL: Signal<CriticalSectionRawMutex, ()> = Signal::new();
313
314/// Stack-format a line with a `[SSSSS.mmm LEVEL ] msg\r\n` header. Formats
315/// WITHOUT the CRLF first, then guarantees termination so an over-long record is
316/// truncated-but-terminated, never merged into the next (which would corrupt the
317/// `[hil-cat]` CSV dump).
318fn format_line(level: &str, args: core::fmt::Arguments<'_>) -> String<LINE_CAP> {
319    let now = embassy_time::Instant::now();
320    let mut line: String<LINE_CAP> = String::new();
321    let _ = write!(
322        line,
323        "[{:05}.{:03} {:<5}] {}",
324        now.as_secs(),
325        now.as_millis() % 1000,
326        level,
327        args
328    );
329    while line.len() + 2 > LINE_CAP {
330        let _ = line.pop();
331    }
332    let _ = line.push_str("\r\n");
333    line
334}
335
336/// Format + push a line into the ring, signal the drain. O(1) on the caller
337/// side: brief CS for the memcpy + an idempotent signal. **Non-blocking** —
338/// on full the oldest bytes are overwritten (protects time-critical producers).
339fn push_line(level: &str, args: core::fmt::Arguments<'_>) {
340    let line = format_line(level, args);
341    RING.lock(|r| r.borrow_mut().write(line.as_bytes()));
342    DRAIN_SIGNAL.signal(());
343}

347/// Filter spec supplied by the application at [`init`] — the console is the
348/// `log` backend and implements the mechanism, the app owns the policy.
349/// `env_logger` syntax: `info,sdspi=debug`; the longest matching
350/// `target=level` directive wins, a bare level is the default.
351///
352/// Held as a raw pointer pair, not behind a mutex: `enabled` runs on EVERY
353/// log macro that passes `log`'s own atomic gate, and taking a critical
354/// section there to read a `&'static str` would cost more than the filtering.
355/// Written once in `init` before any task logs; reads are relaxed and see
356/// either null (pre-init, treated as the default) or the final value.
357static SPEC_PTR: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
358static SPEC_LEN: core::sync::atomic::AtomicUsize = core::sync::atomic::AtomicUsize::new(0);
359/// Level of the spec's bare directive — the common case, answered without
360/// touching the spec string at all.
361static DEFAULT_LEVEL: core::sync::atomic::AtomicU8 = core::sync::atomic::AtomicU8::new(3);
362/// Whether any `target=level` directive exists. When none do (production
363/// `"info"`), `enabled` is one relaxed load and a compare, as it was before
364/// per-module filtering existed.
365static HAS_DIRECTIVES: core::sync::atomic::AtomicBool = core::sync::atomic::AtomicBool::new(false);
366
367fn spec() -> &'static str {
368    use core::sync::atomic::Ordering::Relaxed;
369    let ptr = SPEC_PTR.load(Relaxed) as *const u8;
370    if ptr.is_null() {
371        return "info";
372    }
373    // SAFETY: written once in `init` from a `&'static str`; never mutated.
374    unsafe {
375        core::str::from_utf8_unchecked(core::slice::from_raw_parts(ptr, SPEC_LEN.load(Relaxed)))
376    }
377}
378
379fn level_from_u8(v: u8) -> log::LevelFilter {
380    match v {
381        0 => log::LevelFilter::Off,
382        1 => log::LevelFilter::Error,
383        2 => log::LevelFilter::Warn,
384        3 => log::LevelFilter::Info,
385        4 => log::LevelFilter::Debug,
386        _ => log::LevelFilter::Trace,
387    }
388}
389
390fn level_of(s: &str) -> Option<log::LevelFilter> {
391    match s.as_bytes() {
392        b"off" => Some(log::LevelFilter::Off),
393        b"error" => Some(log::LevelFilter::Error),
394        b"warn" => Some(log::LevelFilter::Warn),
395        b"info" => Some(log::LevelFilter::Info),
396        b"debug" => Some(log::LevelFilter::Debug),
397        b"trace" => Some(log::LevelFilter::Trace),
398        _ => None,
399    }
400}
401
402fn filter_for(target: &str) -> log::LevelFilter {
403    let spec = spec();
404    let mut best = log::LevelFilter::Info;
405    let mut best_len = 0;
406    for directive in spec.split(',') {
407        match directive.split_once('=') {
408            None => {
409                if best_len == 0 {
410                    best = level_of(directive.trim()).unwrap_or(log::LevelFilter::Info);
411                }
412            }
413            Some((prefix, level)) => {
414                let prefix = prefix.trim();
415                if target.starts_with(prefix) && prefix.len() > best_len {
416                    if let Some(l) = level_of(level.trim()) {
417                        best = l;
418                        best_len = prefix.len();
419                    }
420                }
421            }
422        }
423    }
424    best
425}

427impl log::Log for ConsoleLogger {
428    fn enabled(&self, meta: &log::Metadata) -> bool {
429        use core::sync::atomic::Ordering::Relaxed;
430        // Fast path, and the only one production takes: anything at or above
431        // the default level passes without looking at the spec.
432        if meta.level() <= level_from_u8(DEFAULT_LEVEL.load(Relaxed)) {
433            return true;
434        }
435        // Below the default, and no per-target directive can raise it.
436        if !HAS_DIRECTIVES.load(Relaxed) {
437            return false;
438        }
439        meta.level() <= filter_for(meta.target())
440    }
441
442    fn log(&self, record: &log::Record) {
443        if !self.enabled(record.metadata()) {
444            return;
445        }
446        // Format level as a string so the column width matches the prior layout
447        // exactly (existing log scrapers depend on it).
448        let lvl = match record.level() {
449            log::Level::Error => "ERROR",
450            log::Level::Warn => "WARN ",
451            log::Level::Info => "INFO ",
452            log::Level::Debug => "DEBUG",
453            log::Level::Trace => "TRACE",
454        };
455        push_line(lvl, *record.args());
456    }
457
458    fn flush(&self) {}
459}

463/// Register the console as the global `log` backend. Call once, early. The
464/// ring is statically initialised, so producers can write immediately —
465/// pre-[`drain_task`] writes simply accumulate in the ring and are flushed
466/// once the drain runs.
467/// `spec` is an `env_logger`-style filter (`info,sdspi=debug`), typically the
468/// app's `ESP_LOG` read at the APP's compile time — the console previously
469/// hardcoded Info here, which silently discarded every `module=debug`
470/// directive from the day esp-println (whose logger parsed `ESP_LOG`) was
471/// dropped.
472pub fn init(spec: &'static str) {
473    use core::sync::atomic::Ordering::Relaxed;
474    SPEC_LEN.store(spec.len(), Relaxed);
475    SPEC_PTR.store(spec.as_ptr() as usize, Relaxed);
476    let mut default = log::LevelFilter::Info;
477    let mut directives = false;
478    for directive in spec.split(',') {
479        match directive.split_once('=') {
480            None => default = level_of(directive.trim()).unwrap_or(default),
481            Some(_) => directives = true,
482        }
483    }
484    DEFAULT_LEVEL.store(default as u8, Relaxed);
485    HAS_DIRECTIVES.store(directives, Relaxed);
486    // Global gate = the most verbose level any directive asks for; per-target
487    // narrowing happens in `enabled`.
488    let mut max = log::LevelFilter::Info;
489    for directive in spec.split(',') {
490        let level = directive
491            .split_once('=')
492            .map(|(_, l)| l)
493            .unwrap_or(directive);
494        if let Some(l) = level_of(level.trim()) {
495            if l > max {
496                max = l;
497            }
498        }
499    }
500    let _ = log::set_logger(&LOGGER);
501    log::set_max_level(max);
502    // Liveness self-test: one debug!-level line, enabled by a directive in the
503    // HIL builds' filter spec. The debug channel was once dead for 11 weeks
504    // with nothing anywhere failing; the harness asserts this marker at boot,
505    // so a dead filter now fails every HIL run instead of no test ever.
506    log::debug!("{} spec={}", markers::DBG_SELFTEST, spec);
507}

550/// One-call console bring-up: register the `log` backend with `cfg.filter` and,
551/// when `cfg.serial` is `Some`, build the chip's transport + spawn the single
552/// [`drain_task`]. Replaces the hand-wired `init()` + `setup()` + drain-spawn
553/// sequence in every binary. Call once from main (the core that owns the drain;
554/// Fire27's TX `into_async()` binds the IRQ to the calling core).
555///
556/// Returns a [`Console`] whose `rx` (when present) goes to the serial-command
557/// reader so log TX and command RX share the one port (R7).
558pub fn install(spawner: embassy_executor::Spawner, cfg: Config) -> Console {
559    init(cfg.filter);
560    let console = {
561        #[cfg(feature = "console-serial")]
562        {
563            let rx = cfg.serial.map(|res| {
564                let (rx, tx) = install_serial(res);
565                crate::must_spawn!(spawner, drain_task(tx));
566                rx
567            });
568            Console { rx }
569        }
570        // R9 serial-free: backend registered above; no transport, no drain task.
571        #[cfg(not(feature = "console-serial"))]
572        {
573            let _ = &spawner;
574            match cfg.serial {
575                Some(res) => match res {}, // SerialResources is uninhabited here
576                None => {}
577            }
578            Console {}
579        }
580    };
581    // As early as possible: the first BSP-emitted log line, right after the
582    // backend (and transport, if any) are up. See markers::IDENTITY.
583    #[cfg(feature = "app-desc")]
584    crate::log_boot_identity();
585    console
586}

595/// Back-pressuring line sink — the **R10 injection point**. A control crate must
596/// not depend on this BSP directly; instead it defines its own `LineSink`-style
597/// trait and the *binary* injects a ~5-line newtype whose `send_line` forwards
598/// here (the trait stays consumer-side, so neither BSP nor control crate depends
599/// on the other). This is also the bulk-dump path for the HIL `:cat` CSV.
600///
601/// Bulk-dump emit (HIL `:cat` CSV read-back) — **back-pressuring** (lossless).
602/// Unlike the hot log path ([`push_line`], which overwrites-oldest and never
603/// blocks to protect time-critical producers like RWBLE), this AWAITS until the
604/// ring has room for the whole line before writing, then signals the drain. So a
605/// fast `:cat` dump self-paces to the TX drain rate instead of overflowing the
606/// ~4 KB ring and dropping lines (which made `log_interval`'s read-back show
607/// false gaps). Safe because the dump runs on a non-time-critical task and can
608/// tolerate await latency; the `log!()` path does NOT use this.
609pub async fn send_line(args: core::fmt::Arguments<'_>) {
610    let line = format_line("INFO ", args);
611    loop {
612        // Reserve only if the WHOLE line fits — a partial write would still
613        // overwrite-on-full and corrupt the dump. LINE_CAP < RING_SIZE, so it
614        // always fits once the drain has caught up.
615        let wrote = RING.lock(|r| {
616            let mut r = r.borrow_mut();
617            if r.free() >= line.len() {
618                r.write(line.as_bytes());
619                true
620            } else {
621                false
622            }
623        });
624        if wrote {
625            DRAIN_SIGNAL.signal(());
626            return;
627        }
628        // Ring full — wait for the drain to free space, then retry. (No lost
629        // wakeup: SPACE_SIGNAL latches, so a signal between the check above and
630        // this wait still wakes us.)
631        SPACE_SIGNAL.wait().await;
632    }
633}

635/// The single console writer: pulls contiguous slices from the ring and
636/// writes them via the target's async TX sink. `write_all().await` parks on
637/// the TX-done IRQ when the FIFO is full — no spin, no interrupts-off, no
638/// cross-core contention. When the ring is empty, awaits [`DRAIN_SIGNAL`]
639/// (woken by every producer write). Spawn once from the binary's main
640/// (fire27: pass `tx.into_async()`; cores3: the split TX is already async).
641#[cfg(feature = "console-serial")]
642#[embassy_executor::task]
643pub async fn drain_task(mut tx: ConsoleTxAsync<'static>) {
644    // Per-iteration scratch. 256 B on the task stack is fine; the loop runs
645    // again immediately to drain whatever didn't fit.
646    let mut scratch = [0u8; 256];
647    // Overflow-marker rate-limit state (see below).
648    let mut drop_accum: u32 = 0;
649    let mut last_drop_report = embassy_time::Instant::now();
650    loop {
651        // Read the next slice AND any overwrite count in one lock.
652        let (n, dropped) = RING.lock(|r| {
653            let mut r = r.borrow_mut();
654            let n = r.read_and_consume(&mut scratch);
655            (n, r.take_dropped())
656        });
657        drop_accum = drop_accum.saturating_add(dropped);
658        // Surface a ring overrun as a LOUD, greppable marker so dropped bytes
659        // never masquerade as a clean gap (the silent loss that made
660        // `log_interval`'s read-back fail; host can grep `[CONSOLE-DROP`).
661        //
662        // RATE-LIMITED + coalesced: the marker is written by the drain straight
663        // to TX (bypassing the ring, so it costs no ring space and can't itself
664        // be overwritten), but it still shares TX bandwidth + drain time with
665        // the data it's preserving. Emitting it every 256-B chunk under a
666        // sustained log storm would steal drain throughput and AMPLIFY the
667        // overrun (more drops → more markers → slower drain). So emit at most
668        // ~4×/s with the accumulated byte count, plus an immediate flush the
669        // moment the ring drains (episode end). Never blocks/back-pressures
670        // producers — it only delays the drain slightly, now bounded.
671        if drop_accum > 0 {
672            let now = embassy_time::Instant::now();
673            if n == 0 || (now - last_drop_report).as_millis() >= 250 {
674                let mut mark: String<48> = String::new();
675                let _ = write!(mark, "\r\n[CONSOLE-DROP {}B]\r\n", drop_accum);
676                let _ = tx.write_all(mark.as_bytes()).await;
677                drop_accum = 0;
678                last_drop_report = now;
679            }
680        }
681        if n == 0 {
682            DRAIN_SIGNAL.wait().await;
683            continue;
684        }
685        // Freed `n` bytes — wake any back-pressuring producer (the :cat dump).
686        SPACE_SIGNAL.signal(());
687        // Write outside the lock — `.await` is NEVER inside the ring CS.
688        let _ = tx.write_all(&scratch[..n]).await;
689    }
690}

692// ---- RTC-persistent panic breadcrumb (R8) ----
693
694/// Marks a written crumb, distinguishing a real breadcrumb from uninitialised
695/// RTC RAM after a cold boot / power cycle.
696const CRUMB_MAGIC: u32 = 0x6D35_C0DE;
697
698// Breadcrumb word layout in the RTC-fast persistent array.
699const CRUMB_MAGIC_IDX: usize = 0;
700const CRUMB_REASON_IDX: usize = 1;
701const CRUMB_FILE_PTR_IDX: usize = 2;
702const CRUMB_FILE_LEN_IDX: usize = 3;
703
704/// Breadcrumb in RTC-fast RAM as `[magic, reason, file_ptr, file_len]`.
705/// `#[ram(unstable(rtc_fast, persistent))]` keeps it out of the data-init that
706/// runs on a watchdog/software reset, so it survives the RWDT recovery that
707/// follows a panic — exactly the window R8 needs. A `[u32; 4]` (not a struct)
708/// because esp-hal's persistent section requires `Persistable`, which is
709/// implemented for primitive arrays; `usize` is 32-bit on these chips, so the
710/// `.rodata` file-`&str` pointer + length fit in a `u32` each. The pointer is
711/// valid to reconstruct because the crumb only survives a *warm* reset of the
712/// *same* firmware image.
713#[ram(unstable(rtc_fast, persistent))]
714static mut PANIC_CRUMB: [u32; 4] = [0; 4];
715
716/// A panic breadcrumb recovered from the previous run: `location` is the panic
717/// file, `reason` a 32-bit digest of the panic message.
718pub struct PanicCrumb {
719    pub location: &'static str,
720    pub reason: u32,
721}
722
723/// FNV-1a 32-bit digest of the panic message — the R8 "reason-digest".
724fn reason_digest(s: &str) -> u32 {
725    let mut h: u32 = 0x811c_9dc5;
726    for &b in s.as_bytes() {
727        h ^= b as u32;
728        h = h.wrapping_mul(0x0100_0193);
729    }
730    h
731}
732
733/// Record the panic breadcrumb (file location + message digest) in
734/// RTC-persistent RAM before halting. Called by [`on_panic`].
735fn write_panic_crumb(info: &core::panic::PanicInfo<'_>, msg: &str) {
736    let (file_ptr, file_len) = info
737        .location()
738        .map(|l| (l.file().as_ptr() as usize, l.file().len()))
739        .unwrap_or((0, 0));
740    let mut crumb = [0u32; 4];
741    crumb[CRUMB_MAGIC_IDX] = CRUMB_MAGIC;
742    crumb[CRUMB_REASON_IDX] = reason_digest(msg);
743    crumb[CRUMB_FILE_PTR_IDX] = file_ptr as u32;
744    crumb[CRUMB_FILE_LEN_IDX] = file_len as u32;
745    // SAFETY: panic context is single-threaded and terminal; the only other
746    // access is `take_panic_breadcrumb` at boot before any task runs.
747    unsafe { core::ptr::addr_of_mut!(PANIC_CRUMB).write(crumb) };
748}
749
750/// Read and clear the previous run's panic breadcrumb, if any. Call **once** at
751/// boot, before [`install`], and log the result (the [`markers::PREV_PANIC`]
752/// line) — that read-back is the cross-transport fault contract (R8), identical
753/// on both targets. Returns `None` on a clean boot or once the crumb is taken.
754pub fn take_panic_breadcrumb() -> Option<PanicCrumb> {
755    // SAFETY: called once at boot before any task runs — no concurrent access.
756    unsafe {
757        let ptr = core::ptr::addr_of_mut!(PANIC_CRUMB);
758        let c = ptr.read();
759        if c[CRUMB_MAGIC_IDX] != CRUMB_MAGIC {
760            return None;
761        }
762        (*ptr)[CRUMB_MAGIC_IDX] = 0; // clear in place → reported exactly once
763        let file_ptr = c[CRUMB_FILE_PTR_IDX] as usize;
764        let location = if file_ptr != 0 {
765            core::str::from_utf8_unchecked(core::slice::from_raw_parts(
766                file_ptr as *const u8,
767                c[CRUMB_FILE_LEN_IDX] as usize,
768            ))
769        } else {
770            "?"
771        };
772        Some(PanicCrumb {
773            location,
774            reason: c[CRUMB_REASON_IDX],
775        })
776    }
777}

779/// Shared message-only panic handler for both targets. Pushes the panic info
780/// into the ring (alongside the pre-panic context that's already there), then
781/// **synchronously drains the ring** via the raw FIFO poker — the async drain
782/// task is gone (or never started) so it can't service the ring for us. After
783/// the drain, halts so the fault stays visible. No stack walk: message-only
784/// on both targets (deliberate, symmetric), which is why neither `esp-backtrace`
785/// nor `esp-println` is pulled in. The binary's `#[panic_handler]` is a
786/// one-line wrapper around this.
787pub fn on_panic(info: &core::panic::PanicInfo<'_>) -> ! {
788    // Push the panic into the ring. Producer-side is unchanged from any
789    // normal log: brief CS, no await.
790    let mut line: String<256> = String::new();
791    let _ = write!(line, "\r\n[PANIC] {}\r\n", info);
792    // R8: record the breadcrumb (location + message digest) in RTC-persistent
793    // RAM BEFORE any best-effort transport print — the crumb is the contract; a
794    // CDC/UART flush that can't complete post-halt must not gate it.
795    write_panic_crumb(info, line.as_str());
796    RING.lock(|r| r.borrow_mut().write(line.as_bytes()));
797
798    // Best-effort transport print: synchronously drain the ring via
799    // `boot_panic_write` (the async drain task is gone/never-started). Compiled
800    // out in a serial-free build (R9) — the RTC breadcrumb above is the contract;
801    // here we just halt and let the RWDT recover.
802    #[cfg(feature = "console-serial")]
803    loop {
804        let mut chunk = [0u8; 64];
805        let n = RING.lock(|r| r.borrow_mut().read_and_consume(&mut chunk));
806        if n == 0 {
807            break;
808        }
809        boot_panic_write(&chunk[..n]);
810    }
811
812    loop {
813        core::sync::atomic::compiler_fence(core::sync::atomic::Ordering::SeqCst);
814    }
815}
