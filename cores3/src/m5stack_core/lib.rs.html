2//! Board support crate for the **M5Stack Fire27** (ESP32) and **CoreS3**
3//! (ESP32-S3).
4//!
5//! Provides chip-agnostic peripheral drivers ([`driver`]), a shared async I2C
6//! bus and reusable `embassy`-based IO task loops ([`io`]), and board bring-up
7//! helpers ([`board`]).
8//!
9//! The crate also owns the board/chip boilerplate a binary would otherwise
10//! hand-roll, so a consumer's `main` collapses to a thin entry shell:
11//!
12//! - [`mem::init_heap`] — the global heap (esp-alloc DRAM regions + HIL-proven
13//!   per-board sizes; `heap` feature, implied by `psram`).
14//! - [`io::console::install`] — one-call logging over the chip's native
15//!   transport (UART0 / USB-Serial-JTAG CDC) + an RTC panic breadcrumb
16//!   ([`io::console::take_panic_breadcrumb`]) + (`app-desc` feature) the
17//!   firmware identity, logged once as the very first BSP-emitted line.
18//! - the `panic-handler` feature exports the `#[panic_handler]`, and
19//!   [`app_desc!`] the esp-idf app descriptor (`app-desc` feature, implied by
20//!   `heap`) — plus [`app_elf_sha256`], and, under `identity`, an enforced
21//!   build-time git identity in `app_desc!()`'s version field (see the
22//!   README's "Firmware identity" section).
23//! - [`board::run_app_core`] — the second-core harness (`multicore` feature).
24//! - [`io::input_caps`] — the board's input model (keypad vs pointer), so a UI
25//!   installs the matching indev without hardcoding the board.
26//!
27//! Exactly one board feature must be enabled: `fire27` (xtensa-esp32) or
28//! `cores3` (xtensa-esp32s3). Radio (`ble`/`wifi`/`wifi-sta`/`coex`), `heap`/
29//! `psram`/`app-desc`/`identity`, `console-serial`, `panic-handler`, and
30//! `multicore` are orthogonal opt-ins. See the README for the full feature
31//! matrix and usage examples.
32#![no_std]

62/// Replaces embassy-executor's `Spawner::must_spawn`, dropped in 0.10: panics
63/// on pool exhaustion with call-site context. Works for `Spawner` and
64/// `SendSpawner`.
65#[macro_export]
66macro_rules! must_spawn {
67    ($spawner:expr, $task:expr) => {
68        $spawner.spawn(
69            $task.unwrap_or_else(|e| {
70                ::core::panic!(concat!("spawn ", stringify!($task), ": {:?}"), e)
71            }),
72        )
73    };
74}

97// `app_desc!` wraps esp-bootloader-esp-idf's `esp_app_desc!`. The bootloader
98// crate is pulled by the `app-desc` feature (implied by `heap`); re-exported
99// (hidden) so the macro can name it from the call site without the binary
100// depending on it directly.
101#[cfg(feature = "app-desc")]
102#[doc(hidden)]
103pub use esp_bootloader_esp_idf as __bootloader;

105/// Reads back the descriptor [`app_desc!`] emits.
106///
107/// [`app_desc!`] expands to a `static` named `ESP_APP_DESC` **at its call
108/// site** (i.e. in the binary, not this crate), so this reads it back by its
109/// linker symbol (`esp_app_desc`, `EspAppDesc` is `#[repr(C)]`) rather than by
110/// path — the one way a BSP function can reach a descriptor the *consumer*
111/// created. Requires [`app_desc!`] to have been invoked somewhere in the
112/// binary: otherwise this fails to **link**, not silently reads zeroes.
113///
114/// This indirection matters beyond just "how do we reach it": reading
115/// `ESP_APP_DESC` via an `extern` symbol, rather than by path from *inside*
116/// the same crate that defines it, is what keeps [`app_elf_sha256`] correct.
117/// `espflash` patches that field into the flashed image **after**
118/// compilation — the compiler only ever sees the macro's zero initializer.
119/// A same-crate path read of a `static` with a compiler-visible initializer
120/// is free to const-fold to that initializer (needing a `read_volatile` to
121/// stop it); an `extern` read of a symbol whose initializer lives in a
122/// *different* compilation unit has no initializer to see in the first
123/// place, so the hazard cannot arise — no volatile needed. `version` has no
124/// such concern (nothing patches it post-link), but `app_elf_sha256` does:
125/// don't "simplify" this back to a path reference.
126#[cfg(feature = "app-desc")]
127pub fn app_desc() -> &'static __bootloader::EspAppDesc {
128    unsafe extern "C" {
129        #[link_name = "esp_app_desc"]
130        static ESP_APP_DESC: __bootloader::EspAppDesc;
131    }
132    unsafe { &ESP_APP_DESC }
133}

135/// Bytes budget for [`__pkg_version_bytes`] — plenty for any real semver
136/// string (`"1.2.3"` is 5 bytes; `"1.0.0-beta.12"` is 13). Unlike the
137/// identity mark, a too-long crate version here is a cosmetic display
138/// nicety, not an identity-tracking risk, so it's a silent truncation, not
139/// enforced by a compile-time assertion the way `app_desc!`'s mark is.
140#[cfg(feature = "app-desc")]
141#[doc(hidden)]
142pub const PKG_VERSION_BYTES: usize = 16;
143
144/// Zero-pads/truncates `s` into a fixed-size, C-safe byte array — the same
145/// shape `EspAppDesc`'s own fields use, chosen deliberately: a plain `&str`
146/// (a fat pointer) has no stable ABI for an `extern` static to read back
147/// across the crate boundary the way [`app_desc()`] reads `EspAppDesc`
148/// (which is `#[repr(C)]`); a fixed byte array does. Used only by
149/// [`app_desc!`]'s expansion, to export `CARGO_PKG_VERSION` for
150/// [`pkg_version()`] to read back — needed because `identity` repurposes
151/// `EspAppDesc::version` for the git mark, so the crate version isn't
152/// available there anymore.
153#[cfg(feature = "app-desc")]
154#[doc(hidden)]
155pub const fn __str_to_fixed<const N: usize>(s: &str) -> [u8; N] {
156    let bytes = s.as_bytes();
157    let mut out = [0u8; N];
158    let mut i = 0;
159    while i < bytes.len() && i < N {
160        out[i] = bytes[i];
161        i += 1;
162    }
163    out
164}

166/// Reads back the `CARGO_PKG_VERSION` [`app_desc!`] exports alongside the
167/// descriptor — see [`__str_to_fixed`] for why this needs its own static
168/// rather than reusing [`app_desc()`]'s mechanism.
169///
170/// Gated on `identity`, not merely `app-desc`: without `identity` the
171/// descriptor's own `version` field still holds `CARGO_PKG_VERSION`, so
172/// [`log_boot_identity`] reads it straight from there and this reader has no
173/// caller. `app_desc!` exports the symbol in **both** cases, so a consumer that
174/// wants it can still link against it — only this crate's private reader is
175/// conditional.
176#[cfg(all(feature = "app-desc", feature = "identity"))]
177fn pkg_version() -> &'static str {
178    unsafe extern "C" {
179        #[link_name = "m5stack_core_pkg_version"]
180        static PKG_VERSION: [u8; PKG_VERSION_BYTES];
181    }
182    let bytes = unsafe { &PKG_VERSION };
183    let len = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
184    core::str::from_utf8(&bytes[..len]).unwrap_or("?")
185}

187/// The ELF's content hash, straight from the esp-idf application descriptor —
188/// the first bytes distinguish two images built from the same commit with
189/// different uncommitted edits. Unambiguous with no consumer input needed: it
190/// is a function of the linked image, computed and patched in by `espflash`.
191/// Requires [`app_desc!`] to have been invoked (see [`app_desc()`]).
192#[cfg(feature = "app-desc")]
193pub fn app_elf_sha256() -> &'static [u8; 32] {
194    app_desc().app_elf_sha256()
195}

197/// Logs the descriptor's version (plain `CARGO_PKG_VERSION`, or the enforced
198/// `<pkg>/<bin>/<features>/<hash><dirty>` git mark under `identity` — which
199/// already carries the binary name, so it isn't repeated separately there),
200/// the crate's `version=` (always the real `CARGO_PKG_VERSION`, even under
201/// `identity` where the descriptor's own version field no longer holds it —
202/// see [`pkg_version()`]), and a 6-byte `app_elf_sha256` prefix, once, as
203/// early as possible on boot — called by [`io::console::install`], not meant
204/// to be called directly. Like [`app_elf_sha256`], requires [`app_desc!`] to
205/// have been invoked somewhere in the binary (a link error otherwise, not a
206/// silent no-op): any binary enabling `app-desc` — directly, or via `heap` —
207/// is expected to call [`app_desc!`], matching this crate's existing "thin
208/// entry shell" framing.
209#[cfg(feature = "app-desc")]
210pub(crate) fn log_boot_identity() {
211    use core::fmt::Write as _;
212
213    let desc = app_desc();
214    let sha = desc.app_elf_sha256();
215    let mut hex = heapless::String::<12>::new();
216    for byte in &sha[..6] {
217        let _ = write!(hex, "{byte:02x}");
218    }
219    #[cfg(feature = "identity")]
220    log::info!(
221        "{} {} version={} app_elf_sha256={hex}",
222        io::console::markers::IDENTITY,
223        desc.version(),
224        pkg_version()
225    );
226    #[cfg(not(feature = "identity"))]
227    log::info!(
228        "{} {} version={} app_elf_sha256={hex}",
229        io::console::markers::IDENTITY,
230        desc.project_name(),
231        desc.version()
232    );
233}

235/// Emit the esp-idf application descriptor. Invoke once **in the binary** (not
236/// the BSP) so it captures the *application's* `CARGO_PKG_VERSION`. Thin wrapper
237/// over `esp_bootloader_esp_idf::esp_app_desc!` so the binary keeps a single
238/// `m5stack_core::app_desc!();` line instead of naming the bootloader crate.
239///
240/// With the `identity` feature off (default), the descriptor's version field
241/// is plain `CARGO_PKG_VERSION`, as always. With `identity` on, this same call
242/// site — unchanged — instead requires `M5STACK_CORE_BUILD_MARK`, a build-time
243/// git descriptor the consumer's own `build.rs` sets (see the
244/// `m5stack-core-build` crate). BSP owns the mechanism, never the content: if
245/// the `build.rs` wiring is missing, `env!()` fails to compile in the
246/// *consumer's* crate — a real compile error, not a silent fallback.
247///
248/// `project_name` is `CARGO_BIN_NAME`, not `CARGO_PKG_NAME`: a package can
249/// have more than one `[[bin]]`, and only the per-binary compilation (where
250/// this macro expands) knows which one is being built — `CARGO_PKG_NAME`
251/// would report the same value for every binary in a multi-bin package.
252///
253/// Also exports `CARGO_PKG_VERSION` under its own linker symbol (see
254/// [`__str_to_fixed`]) — even though the descriptor's own `version` field
255/// already holds it here, `identity`'s arm below doesn't have that field
256/// free, so both arms export it the same way for [`log_boot_identity`] /
257/// [`pkg_version()`] to read uniformly.
258#[cfg(all(feature = "app-desc", not(feature = "identity")))]
259#[macro_export]
260macro_rules! app_desc {
261    () => {
262        #[unsafe(export_name = "m5stack_core_pkg_version")]
263        #[used]
264        static __M5STACK_CORE_PKG_VERSION: [u8; $crate::PKG_VERSION_BYTES] =
265            $crate::__str_to_fixed(env!("CARGO_PKG_VERSION"));
266        $crate::__bootloader::esp_app_desc!(
267            env!("CARGO_PKG_VERSION"),
268            env!("CARGO_BIN_NAME"),
269            $crate::__bootloader::BUILD_TIME,
270            $crate::__bootloader::BUILD_DATE,
271            $crate::__bootloader::ESP_IDF_COMPATIBLE_VERSION,
272            $crate::__bootloader::MMU_PAGE_SIZE,
273            0,
274            u16::MAX,
275            $crate::__bootloader::SECURE_VERSION
276        );
277    };
278}

280/// See the non-`identity` [`app_desc!`] above — same default call site. The
281/// version field becomes `CARGO_PKG_NAME` + `CARGO_BIN_NAME` joined with the
282/// git mark (`<pkg>/<bin>/<features>/<hash><dirty>`, e.g.
283/// `demos/display/crypto-opt/0f63a4926303+`) — both names included for the
284/// same reason as `project_name` above (a package can have more than one
285/// `[[bin]]`, and disambiguating *which package* matters too once several
286/// crates in a workspace all use `identity`), joined here (rather than by
287/// `m5stack-core-build`) because only this per-binary compilation knows
288/// `CARGO_BIN_NAME`; a `build.rs` runs once per *package* and can't know
289/// which binary it's describing. `CARGO_PKG_VERSION` is exported separately
290/// (see [`__str_to_fixed`]) since the descriptor's `version` field is now
291/// the mark, not the crate version.
292///
293/// `EspAppDesc::version` is a fixed 32-byte C string with no reserved NUL
294/// terminator (see `m5stack-core-build`'s docs) — 31 bytes is the true safe
295/// ceiling, and real package/binary names routinely don't leave room for a
296/// features tag alongside a 12-hex commit (a package/binary pair over ~17
297/// bytes combined already eats the whole budget). Enforced here as a real
298/// compile error, not a silent truncation.
299///
300/// **`app_desc!("prefix")`** — an optional string-literal argument replaces
301/// the automatic `CARGO_PKG_NAME`/`CARGO_BIN_NAME` join with exactly what you
302/// pass, e.g. `app_desc!("oxichg/evcc-hl")`. This is the lever for a project
303/// whose real names don't fit: nothing here truncates or abbreviates on your
304/// behalf (an automatic scheme is a guess, and a wrong guess in an identity
305/// string is worse than no identity — see `#43`'s own reasoning for not
306/// letting the BSP derive a commit itself), so *you* decide what the prefix
307/// says and how long it is. `project_name` still reports the real
308/// `CARGO_BIN_NAME` regardless — only the mark's prefix is overridable.
309/// Default (no argument) is unaffected and unchanged for every existing
310/// caller.
311#[cfg(all(feature = "app-desc", feature = "identity"))]
312#[macro_export]
313macro_rules! app_desc {
314    () => {
315        const _: () = ::core::assert!(
316            ::core::concat!(
317                ::core::env!("CARGO_PKG_NAME"), "/", ::core::env!("CARGO_BIN_NAME"), "/",
318                ::core::env!("M5STACK_CORE_BUILD_MARK")
319            ).len() <= 31,
320            "m5stack_core::app_desc!(): the identity mark (CARGO_PKG_NAME + '/' + CARGO_BIN_NAME \
321             + '/' + M5STACK_CORE_BUILD_MARK) exceeds 31 bytes (EspAppDesc::version's safe \
322             ceiling — see m5stack-core-build's docs). Shorten the `features` tag passed to \
323             emit_identity_env(), or pass a shorter prefix explicitly: \
324             app_desc!(\"short-pkg/short-bin\") instead of the default \
325             CARGO_PKG_NAME/CARGO_BIN_NAME."
326        );
327        #[unsafe(export_name = "m5stack_core_pkg_version")]
328        #[used]
329        static __M5STACK_CORE_PKG_VERSION: [u8; $crate::PKG_VERSION_BYTES] =
330            $crate::__str_to_fixed(env!("CARGO_PKG_VERSION"));
331        $crate::__bootloader::esp_app_desc!(
332            ::core::concat!(
333                ::core::env!("CARGO_PKG_NAME"), "/", ::core::env!("CARGO_BIN_NAME"), "/",
334                ::core::env!("M5STACK_CORE_BUILD_MARK")
335            ),
336            env!("CARGO_BIN_NAME"),
337            $crate::__bootloader::BUILD_TIME,
338            $crate::__bootloader::BUILD_DATE,
339            $crate::__bootloader::ESP_IDF_COMPATIBLE_VERSION,
340            $crate::__bootloader::MMU_PAGE_SIZE,
341            0,
342            u16::MAX,
343            $crate::__bootloader::SECURE_VERSION
344        );
345    };
346    ($prefix:literal) => {
347        const _: () = ::core::assert!(
348            ::core::concat!($prefix, "/", ::core::env!("M5STACK_CORE_BUILD_MARK")).len() <= 31,
349            "m5stack_core::app_desc!(\"prefix\"): the identity mark (prefix + '/' + \
350             M5STACK_CORE_BUILD_MARK) exceeds 31 bytes (EspAppDesc::version's safe ceiling — see \
351             m5stack-core-build's docs). Shorten the `features` tag passed to \
352             emit_identity_env(), or the prefix passed here."
353        );
354        #[unsafe(export_name = "m5stack_core_pkg_version")]
355        #[used]
356        static __M5STACK_CORE_PKG_VERSION: [u8; $crate::PKG_VERSION_BYTES] =
357            $crate::__str_to_fixed(env!("CARGO_PKG_VERSION"));
358        $crate::__bootloader::esp_app_desc!(
359            ::core::concat!($prefix, "/", ::core::env!("M5STACK_CORE_BUILD_MARK")),
360            env!("CARGO_BIN_NAME"),
361            $crate::__bootloader::BUILD_TIME,
362            $crate::__bootloader::BUILD_DATE,
363            $crate::__bootloader::ESP_IDF_COMPATIBLE_VERSION,
364            $crate::__bootloader::MMU_PAGE_SIZE,
365            0,
366            u16::MAX,
367            $crate::__bootloader::SECURE_VERSION
368        );
369    };
